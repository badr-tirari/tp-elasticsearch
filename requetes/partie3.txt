# ---------- Exercice 3.2 — match contre term ----------
GET offres/_search
{ "query": { "match": { "description": "projets bancaires" } } }

# Renvoient 0 résultat :
GET offres/_search
{ "query": { "term": { "ville": "paris" } } }

GET offres/_search
{ "query": { "term": { "titre": "Data Engineer Senior" } } }

# Corrections : valeur exacte (casse respectée) et sous-champ keyword "brut"
GET offres/_search
{ "query": { "term": { "ville": "Paris" } } }

GET offres/_search
{ "query": { "term": { "titre.brut": "Data Engineer Senior" } } }

# Pour comprendre : tokens produits pour le champ titre
GET offres/_analyze
{ "field": "titre", "text": "Data Engineer Senior" }

# match avec opérateur AND : tous les termes doivent être présents
GET offres/_search
{
  "query": {
    "match": { "description": { "query": "projets bancaires", "operator": "and" } }
  }
}

# ---------- Exercice 3.3 — multi_match, fuzziness, pondération ----------
GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre", "competences.texte", "description"]
    }
  }
}

GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre", "competences.texte", "description"],
      "fuzziness": "AUTO"
    }
  }
}

GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetis terraform",
      "fields": ["titre^3", "competences.texte", "description"],
      "fuzziness": "AUTO"
    }
  }
}

# Effet visible du poids avec un terme présent dans les titres
GET offres/_search
{
  "query": {
    "multi_match": {
      "query": "cloud terraform",
      "fields": ["titre^3", "competences.texte", "description"],
      "fuzziness": "AUTO"
    }
  }
}

# ---------- Exercice 3.4 — Requête bool ----------
GET offres/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "données", "fields": ["titre", "description"] } }
      ],
      "filter": [
        { "term":  { "contrat": "CDI" } },
        { "terms": { "ville": ["Montpellier", "Toulouse"] } },
        { "range": { "salaire_max": { "gte": 50000 } } }
      ],
      "must_not": [
        { "term": { "teletravail": "aucun" } }
      ],
      "should": [
        { "term": { "competences": "Elasticsearch" } }
      ]
    }
  }
}

# Même requête sans le bloc should (comparaison des _score)
GET offres/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "données", "fields": ["titre", "description"] } }
      ],
      "filter": [
        { "term":  { "contrat": "CDI" } },
        { "terms": { "ville": ["Montpellier", "Toulouse"] } },
        { "range": { "salaire_max": { "gte": 50000 } } }
      ],
      "must_not": [
        { "term": { "teletravail": "aucun" } }
      ]
    }
  }
}

# ---------- Exercice 3.5 — Recherche géographique ----------
